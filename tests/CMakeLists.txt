function(add_telemetry_test name)
    qt_add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE ${ARGN} Qt6::Test)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

add_telemetry_test(tst_telemetrypacket telemetry_common)
add_telemetry_test(tst_sequencetracker telemetry_common)
add_telemetry_test(tst_channelmodel telemetry_monitor_lib)
add_telemetry_test(tst_eventlogmodel telemetry_monitor_lib)
add_telemetry_test(tst_udpreceiver telemetry_monitor_lib)
add_telemetry_test(tst_telemetrycontroller telemetry_monitor_lib)
